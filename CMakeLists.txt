cmake_minimum_required(VERSION 3.24)
project(exactsample LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# sm_86 and sm_89: the architectures the tests have run on (an A10 vGPU and an RTX 4090).
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 86 89)
endif()
option(EXACTSAMPLE_STANDALONE "Compile the CUDA core without PyTorch (no Python operator)" OFF)
find_package(CUDAToolkit REQUIRED)
add_library(exactsample_core STATIC csrc/verify/verify.cu csrc/verify/fused.cu csrc/verify/fused_f32.cu
  csrc/verify/fused_f64.cu csrc/verify/fused_f16.cu csrc/verify/fused_bf16.cu)
target_include_directories(exactsample_core PUBLIC include)
target_link_libraries(exactsample_core PUBLIC CUDA::cudart)
target_compile_options(exactsample_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
if(NOT EXACTSAMPLE_STANDALONE)
  # PyTorch's CMake scripts use their own architecture selector and otherwise
  # autodetect obsolete architectures when the driver is unavailable at build.
  set(TORCH_CUDA_ARCH_LIST "8.6;8.9" CACHE STRING "PyTorch CUDA architectures")
  find_package(Torch REQUIRED)
  if(NOT TARGET torch_cuda)
    message(FATAL_ERROR "CUDA-enabled PyTorch is required for exactsample_cuda. Use EXACTSAMPLE_STANDALONE=ON for offline core compilation.")
  endif()
  add_library(exactsample_cuda SHARED csrc/bindings.cpp)
  target_include_directories(exactsample_cuda PRIVATE ${TORCH_INCLUDE_DIRS})
  target_link_libraries(exactsample_cuda PRIVATE exactsample_core ${TORCH_LIBRARIES} torch_cuda c10_cuda)
  separate_arguments(EXACTSAMPLE_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
  target_compile_options(exactsample_cuda PRIVATE ${EXACTSAMPLE_TORCH_FLAGS})
  set_target_properties(exactsample_cuda PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "$ORIGIN/../torch/lib")
  install(TARGETS exactsample_cuda LIBRARY DESTINATION exactsample)
endif()
